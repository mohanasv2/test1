menuentry "My ISO" {
  set ISOFile="/opt/Live-ISOs/clonezilla-live-20170905-zesty-amd64.iso"
  loopback loop (hd2,gpt2)$ISOFile
  linuxefi (loop)/live/vmlinuz boot=live components config findiso=$ISOFile ip=frommedia toram=filesystem.squashfs union=overlay
  initrdefi (loop)/live/initrd.img
}

menuentry "cloudos92 ISO" {
  insmod ext2
  set isofile="/iso/OpenCloudOS-9.2-20250221.0-x86_64-minimal.iso"
  loopback loop (hd7,5)$isofile
  linux (loop)/images/pxeboot/vmlinuz boot=images iso-scan/filename=$isofile quiet noeject noprompt splash
  initrd (loop)/images/pxeboot/initrd.img
}
